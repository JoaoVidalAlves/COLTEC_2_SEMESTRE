#Atividade 5.1
#Escreva subconsultas escalares para:

#a) listar produtos acima da média dos preços;
SELECT nome_produto, preco FROM produtos WHERE preco > (SELECT AVG(preco) FROM produtos) ORDER BY preco DESC;

#b) encontrar o produto de menor preço;
SELECT nome_produto, preco FROM produtos WHERE preco = (SELECT MIN(preco) FROM produtos);

#c) encontrar vendas realizadas na data mais recente;
SELECT id_venda, data_venda FROM vendas WHERE data_venda >= (SELECT MAX(data_venda) FROM vendas);

#d) listar produtos com custo acima do custo médio;
SELECT nome_produto, custo_aquisicao FROM produtos WHERE custo_aquisicao > ( SELECT AVG(custo_aquisicao) FROM produtos);

#e) encontrar produtos cuja margem preco - custo_aquisicao esteja acima da margem média.
SELECT nome_produto, preco - custo_aquisicao AS margem FROM produtos WHERE preco - custo_aquisicao > (SELECT AVG(preco - custo_aquisicao) FROM produtos) ORDER BY nome_produto;
